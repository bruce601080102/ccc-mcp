+def bucket_for_offset(offset, width):
+    if offset < 0 or width <= 0:
+        raise ValueError('offset and width must be valid')
+    bucket = 0
+    boundary = width
+    while offset > boundary:
+        bucket += 1
+        boundary += width
+    return bucket
